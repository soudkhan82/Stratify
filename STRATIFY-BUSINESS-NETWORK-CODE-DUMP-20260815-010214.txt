function normalizeSeries(raw: unknown): Array<[number, number]> {
  if (!Array.isArray(raw) || raw.length === 0) {
    return [];
  }

  const first = raw[0];

  if (Array.isArray(first)) {
    const nested: Array<[number, number]> = [];

    for (const entry of raw) {
      if (!Array.isArray(entry) || entry.length < 2) {
        continue;
      }

      const seriesYear = Number(entry[0]);
      const seriesValue = Number(entry[1]);

      if (
        Number.isFinite(seriesYear) &&
        Number.isFinite(seriesValue)
      ) {
        nested.push([seriesYear, seriesValue]);
      }
    }

    return nested;
  }

  // Compatibility with any older flattened cache:
  // [1992, value, 1993, value, ...]
  const flat: Array<[number, number]> = [];

  for (let index = 0; index + 1 < raw.length; index += 2) {
    const seriesYear = Number(raw[index]);
    const seriesValue = Number(raw[index + 1]);

    if (
      Number.isFinite(seriesYear) &&
      Number.isFinite(seriesValue)
    ) {
      flat.push([seriesYear, seriesValue]);
    }
  }

  return flat;
}

function formatNumber(
  value: number,
  maximumFractionDigits = 1,
) {
  if (!Number.isFinite(value)) {
    return "n/a";
  }

  return new Intl.NumberFormat("en", {
    maximumFractionDigits,
  }).format(value);
}

function compact(value: number) {
  if (!Number.isFinite(value)) {
    return "n/a";
  }

  return new Intl.NumberFormat("en", {
    notation: "compact",
    maximumFractionDigits: 2,
  }).format(value);
}

  label,
  value,
  onChange,
  children,
}: {
  label: string;
  value: string | number;
  onChange: (value: string) => void;
  children: React.ReactNode;
}) {
  return (
    <label className="min-w-[170px] flex-1">
      <span className="mb-1 block text-[10px] font-black uppercase tracking-[0.12em] text-slate-400">
        {label}

      <select

  );
}

export default function AgriculturePage() {
  const [meta, setMeta] = useState<MetaPayload | null>(null);
  const [crop, setCrop] = useState("wheat");
  const [year, setYear] = useState(2024);
  const [cropData, setCropData] = useState<CropPayload | null>(
    null,
  );
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [selectedIso3, setSelectedIso3] = useState<
    string | null
  >(null);
  const [countryQuery, setCountryQuery] = useState("");

  useEffect(() => {
    let active = true;

    fetch(
      `/data/agriculture/meta.json?v=catalogue-clean`,
      {
        cache: "no-store",
      },
    )
      .then(async (response) => {
        if (!response.ok) {
          throw new Error(
            `Agriculture metadata HTTP ${response.status}`,
          );
        }

        return (await response.json()) as MetaPayload;
      })
      .then((payload) => {
        if (!active) {
          return;
        }

        setMeta(payload);

        if (payload.latestYear) {
          setYear(payload.latestYear);
        }

        // Keep wheat when available; otherwise use first catalogue item.
        if (
          !payload.crops.some((item) => item.key === crop) &&
          payload.crops.length > 0
        ) {
          setCrop(payload.crops[0].key);
        }
      })
      .catch((err: unknown) => {
        if (!active) {
          return;
        }

        setError(
          err instanceof Error
            ? err.message
            : "Unable to load Agriculture Atlas metadata.",
        );
      });

    return () => {
      active = false;
    };
  }, []);

  useEffect(() => {
    if (!meta) {
      return;
    }

    let active = true;

    const dataVersion =
      meta.generatedAt || "catalogue-clean";

    const cacheKey =
      `stratify:agriculture:static:${dataVersion}:${crop}`;

    setLoading(true);
    setError("");

    try {
      const cached = sessionStorage.getItem(cacheKey);

      if (cached) {
        const parsed = JSON.parse(cached) as CropPayload;

        if (
          parsed?.cropKey === crop &&
          Array.isArray(parsed.countries)
        ) {
          setCropData(parsed);
          setLoading(false);

          if (year > parsed.latestYear) {
            setYear(parsed.latestYear);
          }

          return () => {
            active = false;
          };
        }
      }
    } catch {
      // Local static fetch remains the fallback.
    }

    fetch(
      `/data/agriculture/${crop}.json?v=${encodeURIComponent(
        dataVersion,
      )}`,
      {
        cache: "no-cache",
      },
    )
      .then(async (response) => {
        if (!response.ok) {
          throw new Error(
            `Local crop dataset HTTP ${response.status}: ${crop}`,
          );
        }

        return (await response.json()) as CropPayload;
      })
      .then((payload) => {
        if (!active) {
          return;
        }

        setCropData(payload);

        if (year > payload.latestYear) {
          setYear(payload.latestYear);
        }

        try {
          sessionStorage.setItem(
            cacheKey,
            JSON.stringify(payload),
          );
        } catch {
          // Browser storage is optional.
        }
      })
      .catch((err: unknown) => {
        if (!active) {
          return;
        }

        setCropData(null);

        setError(
          err instanceof Error
            ? err.message
            : "Unable to load local crop data.",
        );
      })
      .finally(() => {
        if (active) {
          setLoading(false);
        }
      });

    return () => {
      active = false;
    };
  }, [crop, meta?.generatedAt]);

  const rows = useMemo((): AgricultureMapRow[] => {
    if (!cropData) {
      return [];
    }

    const base = cropData.countries
      .map((country) => {
        const normalizedSeries =
          normalizeSeries(country.series);

        const observation = normalizedSeries.find(
          ([seriesYear]) => seriesYear === year,
        );

        if (!observation) {
          return null;
        }

        const value = Number(observation[1]);

        if (!Number.isFinite(value)) {
          return null;
        }

        return {
          iso3: country.iso3,
          areaCode: country.m49,
          country: country.country,
          lat: country.lat,
          lng: country.lng,
          crop: cropData.crop,
          cropKey: cropData.cropKey,
          year,
          value,
          unit: country.unit || "t",
          rank: null,
          sharePct: null,
        } satisfies AgricultureMapRow;
      })
      .filter(
        (row): row is AgricultureMapRow => row !== null,
      );

    base.sort((a, b) => b.value - a.value);

    const globalTotal = base.reduce(
      (sum, row) => sum + row.value,
      0,
    );

    return base.map((row, index) => ({
      ...row,
      rank: index + 1,
      sharePct:
        globalTotal > 0
          ? (row.value / globalTotal) * 100
          : null,
    }));
  }, [cropData, year]);

  const globalTotal = useMemo(
    () =>
      rows.reduce(
        (sum, row) => sum + row.value,
        0,
      ),
    [rows],
  );

  const selected = useMemo(
    () =>
      rows.find(
        (row) => row.iso3 === selectedIso3,
      ) ?? null,
    [rows, selectedIso3],
  );

  // Never auto-select the top producer.
  // Initial map therefore remains a global distribution view.
  useEffect(() => {
    setSelectedIso3((current) => {
      if (
        current &&
        rows.some((row) => row.iso3 === current)

        return current;
      }

      return null;
    });
  }, [rows]);

  const trend = useMemo((): TrendPoint[] => {
    if (!cropData || !selectedIso3) {

    }

    const country = cropData.countries.find(
      (item) => item.iso3 === selectedIso3,

  function selectRow(row: AgricultureMapRow) {

  return (
    <main className="min-h-screen bg-slate-100">
      <div className="mx-auto w-full max-w-[1480px] px-4 py-5 sm:px-6 lg:px-8">

          <div className="flex flex-col gap-4 xl:flex-row xl:items-end xl:justify-between">
            <div className="max-w-3xl">

                        {item.label}

                  </optgroup>

const WORLD_MAP_URL = "/maps/world-110m.min.geojson";

function compact(value: number) {
  if (!Number.isFinite(value)) return "n/a";

  return new Intl.NumberFormat("en", {
    notation: "compact",
    maximumFractionDigits: 2,
  }).format(value);
}

function BaseWorldLayer() {
  const [world, setWorld] = useState<any>(null);

  useEffect(() => {
    let alive = true;
    const controller = new AbortController();

    fetch(WORLD_MAP_URL, {
      signal: controller.signal,
      cache: "force-cache",
    })
      .then((response) => {
        if (!response.ok) {
          throw new Error(`World map HTTP ${response.status}`);
        }

        return response.json();
      })
      .then((data) => {
        if (alive) setWorld(data);
      })
      .catch((error) => {
        if (error?.name !== "AbortError") {
          console.warn(
            "Agriculture base world layer unavailable:",
            error,
          );
        }
      });

    return () => {
      alive = false;
      controller.abort();
    };
  }, []);

  if (!world) return null;

  return (
    <GeoJSON
      data={world}
      interactive={false}
      style={{
        color: "#b9c8d8",
        weight: 0.65,
        opacity: 0.95,
        fillColor: "#f8fafc",
        fillOpacity: 1,
      }}
    />
  );
}

function MapViewport({
  rows,
  selected,
}: {
  rows: AgricultureMapRow[];
  selected: AgricultureMapRow | null;
}) {
  const map = useMap();

  useEffect(() => {
    if (selected) {
      map.setView(
        [selected.lat, selected.lng],
        Math.max(map.getZoom(), 4),
        { animate: false },
      );
      return;
    }

    if (!rows.length) {
      map.setView([18, 5], 1.5, { animate: false });
      return;
    }

    const points = rows
      .filter(
        (row) =>
          Number.isFinite(row.lat) &&
          Number.isFinite(row.lng),
      )
      .map(
        (row) =>
          [row.lat, row.lng] as [number, number],
      );

    if (!points.length) {
      map.setView([18, 5], 1.5, { animate: false });
      return;
    }

    map.fitBounds(points, {
      paddingTopLeft: [70, 55],
      paddingBottomRight: [70, 55],
      maxZoom: 1.5,
      animate: false,
    });
  }, [map, rows, selected]);

  return null;
}

  let _client: SupabaseClient | null = null;

  function createSupabaseClient(): SupabaseClient {
    const { url, anon } = getEnv();

    if (!url || !anon) {
      throw new Error(
        "Missing NEXT_PUBLIC_SUPABASE_URL or NEXT_PUBLIC_SUPABASE_ANON_KEY. " +
          "Set these in Vercel Project Settings â†’ Environment Variables."
      );
    }

    return createClient(url, anon);
  }

  /**
   * Default export is a Proxy so existing imports keep working:
   *   import supabase from "@/app/config/supabase-config";
   *
   * The proxy defers env validation until first actual usage.
   */
  const supabase: SupabaseClient = new Proxy({} as SupabaseClient, {
    get(_target, prop) {
      if (!_client) _client = createSupabaseClient();
      // eslint-disable-next-line @typescript-eslint/no-unsafe-return

    },
  });

import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";

function copyResponseCookies(
  source: NextResponse,
  destination: NextResponse,
) {
  source.cookies.getAll().forEach(({ name, value, ...options }) => {
    destination.cookies.set(name, value, options);
  });
}

function hasSupabaseSessionCookie(request: NextRequest) {
  return request.cookies
    .getAll()
    .some(
      ({ name }) =>
        name.startsWith("sb-") &&
        (name.includes("auth-token") || name.includes("refresh-token")),
    );
}

export async function updateSession(request: NextRequest) {
  let supabaseResponse = NextResponse.next({
    request,
  });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll();
        },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value }) => {
            request.cookies.set(name, value);
          });

          supabaseResponse = NextResponse.next({
            request,
          });

          cookiesToSet.forEach(({ name, value, options }) => {
            supabaseResponse.cookies.set(name, value, options);
          });
        },
      },
    },
  );

  const pathname = request.nextUrl.pathname;
  const authCode = request.nextUrl.searchParams.get("code");

  // Preserve the OAuth callback hand-off if a provider returns the code
  // to another page unexpectedly.
  if (authCode && pathname !== "/auth/callback") {
    const callbackUrl = request.nextUrl.clone();
    callbackUrl.pathname = "/auth/callback";
    callbackUrl.search = "";
    callbackUrl.searchParams.set("code", authCode);

    const next = request.nextUrl.searchParams.get("next");
    if (next && next.startsWith("/") && !next.startsWith("//")) {
      callbackUrl.searchParams.set("next", next);
    }

    const callbackResponse = NextResponse.redirect(callbackUrl);
    copyResponseCookies(supabaseResponse, callbackResponse);
    return callbackResponse;
  }

  // Authentication is OPTIONAL.
  // Public visitors are never redirected to /login.
  //
  // Only refresh/validate a Supabase session when a session cookie exists.
  // This avoids an unnecessary auth round-trip for anonymous visitors and
  // keeps public browsing fast.
  if (hasSupabaseSessionCookie(request)) {
    try {
      await supabase.auth.getUser();
    } catch (error) {
      console.error("Optional Supabase session refresh failed:", error);
    }
  }

  return supabaseResponse;
}
